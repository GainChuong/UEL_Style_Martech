Hướng Dẫn Kết Nối Database Excel / Google Sheets

Để form trên Landing Page tự động lưu thông tin khách hàng vào file Excel / Google Sheets theo thời gian thực (hoàn toàn miễn phí 100%), bạn thực hiện 3 bước sau:

Bước 1: Tạo Google Sheet mới

Mở trình duyệt vào sheets.new để tạo một bảng tính mới (đặt tên ví dụ: UEL_STYLE_Leads_Database).

Bước 2: Mở Apps Script & Dán mã xử lý tự động

Trên thanh menu của Google Sheet, bấm Tiện ích mở rộng (Extensions) → chọn Apps Script. Xóa mã mặc định và dán đoạn mã bên dưới vào:

// Mã Google Apps Script lưu Form UEL STYLE vào Google Sheet / Excel
// Chuẩn hóa 22 cột theo Table 2 & Table 3 Báo Cáo MarTech (Đa đối tượng & First/Last Touch)
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    var headers = [
      "Mã Lead (lead_id)",
      "Họ và Tên",
      "Email Khách Hàng",
      "Nhóm Đối Tượng (Customer Segment)",
      "Giai Đoạn Hồ Sơ (profile_status)",
      "Xác Thực OTP (otp_status)",
      "Đồng Ý Marketing (consent_marketing)",
      "Sở Thích Sản Phẩm (interests)",
      "Gu Phong Cách (style_preference)",
      "Mức Giá Mong Muốn (price_range)",
      "First-Touch Platform",
      "First-Touch Medium",
      "First-Touch Placement (feed/story/reels/profile/post)",
      "First-Touch Campaign & Creative",
      "Last-Touch Platform",
      "Last-Touch Medium",
      "Last-Touch Placement",
      "Last-Touch Campaign & Creative",
      "Phương Thức Nguồn (source_method)",
      "Độ Tin Cậy Nguồn (confidence)",
      "Session ID",
      "Thời Gian Đăng Ký (submitted_at)"
    ];
    
    // Tạo hàng tiêu đề màu xanh UEL STYLE nếu sheet còn trống
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#2C529D").setFontColor("#FFFFFF");
    }
    
    var email = (data.normalized_email || data.email || "").toString().trim().toLowerCase();
    var rows = sheet.getDataRange().getValues();
    var rowIndex = -1;
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][2] && rows[i][2].toString().trim().toLowerCase() === email) {
        rowIndex = i + 1;
        break;
      }
    }
    
    var rowData = [
      data.lead_id || ("lead-" + new Date().getTime()),
      data.display_name || "",
      email,
      data.customer_segment || data.relation_to_uel || "Khách hàng đại chúng",
      data.profile_status || data.profile_stage || "Stage 1: Đã xác thực OTP",
      data.otp_status || data.email_verified_at || "VERIFIED (Đã xác thực)",
      data.consent_marketing || data.marketing_email_opt_in || "CÓ (Đồng ý)",
      Array.isArray(data.interests) ? data.interests.join(", ") : (data.interests || "Chưa cung cấp"),
      data.style_preference || "Năng động, Streetwear",
      data.price_range || "200k_350k",
      data.first_touch_platform || "facebook",
      data.first_touch_medium || "paid_social",
      data.first_touch_placement || "feed",
      data.first_touch_creative || data.first_touch_campaign || "feed_shirt_a",
      data.last_touch_platform || data.first_touch_platform || "facebook",
      data.last_touch_medium || data.first_touch_medium || "paid_social",
      data.last_touch_placement || data.first_touch_placement || "feed",
      data.last_touch_creative || data.first_touch_creative || "feed_shirt_a",
      data.source_method || "utm_manual",
      data.source_confidence || "high",
      data.session_id || data.anonymous_session_id || ("sess_" + Math.random().toString(36).substr(2, 8)),
      data.submitted_at || new Date().toLocaleString("vi-VN")
    ];
    
    if (rowIndex > 0) {
      sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]);
    } else {
      sheet.appendRow(rowData);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Đã lưu bản ghi chuẩn MarTech vào Google Sheet!" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Database is active" })).setMimeType(ContentService.MimeType.JSON);
}
Bước 3: Triển khai ứng dụng Web & Lấy link

Bấm nút Triển khai (Deploy) ở góc trên bên phải → chọn Tùy chọn triển khai mới (New deployment) → Bấm vào icon bánh răng chọn Ứng dụng web (Web app):

  • Phần Thực thi dưới dạng (Execute as): Chọn Tôi (Email của bạn)
  • Phần Ai có quyền truy cập (Who has access): Chọn Bất kỳ ai (Anyone) (để form web có thể gửi dữ liệu)

Bấm Triển khai, cấp quyền nếu được hỏi, và Sao chép URL ứng dụng web (có dạng https://script.google.com/macros/s/.../exec) dán vào ô Webhook URL trên Admin này.
